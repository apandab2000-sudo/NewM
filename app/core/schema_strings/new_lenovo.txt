TABLE **Lenovo_data_revised** - Always to be used unless User question is about l0_name or l1_name.
(
	geo VARCHAR - Geographical region. UNIQUE VALUES (EMEA, LA, AP, NA), 
	region VARCHAR - Geographical sub region, 
	country VARCHAR , 
	bu VARCHAR - Business Unit. UNIQUE VALUES (IDG, ISG, MBG), 
	bp_type VARCHAR - Type of business partner. UNIQUE VALUES ('Distributor','LNVO','NSP','RELM','Reseller','Retailer','TRAN'), 
	idg_level VARCHAR - IDG Level (Intelligent_Device_Group), 
	dcg_level VARCHAR - DCG/ISG Level (Infrastructure_Solutions_Group), 
	year INTEGER - UNIQUE VALUES (2022, 2023, 2024) 
	segment VARCHAR - Product segment, 
	sub_segment VARCHAR - Product sub-segment, 
	brand VARCHAR - Brand Type, 
	product_group VARCHAR - UNIQUE VALUES ('AR/VR','Client Edge','Commercial All-in-One','Commercial Desktop','Commercial Notebook','Commercial Tablet','Consumer All-in-One','Consumer Desktop','Consumer Notebook','Consumer S&P','Consumer Tablet & Smart Device','Enterprise Product Group','Mobile Internet Digital Home','PC Peripherals','Phone','QDI  Products','Server and Storage S&P','Services','Smart Office','SMB','Notebook','Think Peripherals','Visuals','Wearable','Workstation'), 
	Product_Group_Comb VARCHAR - Product group combination, 
	product_brand VARCHAR , 
	product_series VARCHAR , 
	Product_Series_Comb VARCHAR - Product series combination, 
	global_l0 VARCHAR - Program Type Gross To Net (GTN) global L0, 
	global_l1 VARCHAR - Program Type Gross To Net (GTN) global L1, 
	global_l2 VARCHAR - Program Type Gross To Net (GTN) global L2, 
	top_seller VARCHAR - Indicator for top-selling products, 
	si_type VARCHAR , 
	E_SI FLOAT - Expense Sell in, 
	SI_Gross_Revenue FLOAT - Sell In Gross Revenue, 
	SI_Net_Revenue FLOAT - Sell In Net Revenue, 
	payment_usd_fin FLOAT , 
	payment_usd FLOAT - Rebate GTN Payment USD, 
	GTN FLOAT - Gross To Net (GTN), 
	ST_Revenue FLOAT - Sell Through Revenue, 
	BOX INTEGER - Quantity Unit, 
	gross_profit FLOAT , 
	gross_profit_revenue FLOAT - Gross Profit Revenue. This is used as denominator for GP%, 
	budget_usd FLOAT , 
	D_Inv_Aged_60 INTEGER , 
	R_Inv_Aged_60 INTEGER , 
	Inv_Aged_60 INTEGER - Inventory Aged 60, 
	D_OH_Inv INTEGER , 
	R_OH_Inv INTEGER , 
	OH_Inv INTEGER - On Hand Inventory, 
	gtn_tact FLOAT - Tactical Gross To Net (GTN), 
	gtn_contr FLOAT - Contractual Gross To Net (GTN), 
	budget_tact FLOAT , 
	budget_contr FLOAT , 
	fiscal_year_quarter_end DATE - Fiscal year and quarter (Eg., 2023-10-31)
)
