Table nlv_parent_line(product_code VARCHAR(20),
seasonality VARCHAR(50),
image VARCHAR(10),
sizing JSON (Array of values),
status VARCHAR(10) UNIQUE VALUES ('ADD'),
notes VARCHAR(200),
keylooks VARCHAR(20) UNIQUE VALUES ('T2 KEY LOOK, 'T3 KEY LOOK, 'KEY LOOK'),
first_delivery_date DATE,
last_delivery_date DATE,
season DATE,
ranking FLOAT,
newness VARCHAR(30) UNIQUE VALUES ('NEW', 'C/O', 'EARLY SALES ONLY', 'PULL FORWARD', 'MAY D0'),
vlp_order FLOAT,
segmentation VARCHAR(50),
stories VARCHAR(500),
mandatory VARCHAR(10) UNIQUE VALUES ('No'),
image_url VARCHAR(30),
sustainability_story VARCHAR(100),
replen VARCHAR(10) UNIQUE VALUES ('No', 'Yes'),
distribution VARCHAR(100))
Foreign Key: nlv_parent_line(product_code) references nlv_products(product_code)

Table nlv_products(product_code VARCHAR(20),
title VARCHAR(200),
finish VARCHAR(200),
color VARCHAR(50) UNIQUE VALUES ('BLUE', 'BLACK', 'WHITE', ...),
gender VARCHAR(10) UNIQUE VALUES ('Male', 'Female'),
category VARCHAR(20) UNIQUE VALUES ('TOPS', 'BOTTOMS', 'ACCESSORIES', 'FOOTWEAR'),
sub_category VARCHAR(50),
mfp_category VARCHAR(50),
fit VARCHAR(50),
fabric_innovation VARCHAR(50))
Primary Key: nlv_products(product_code)

Table nlv_customers(key_account_manager VARCHAR(50),
customer_id_bkp VARCHAR(200),
customer_name VARCHAR(200),
customer_id INTEGER,
customer_region VARCHAR(10) UNIQUE VALUES ('LSE', 'NEU'))
Primary Key: nlv_customers(customer_id)

Table nlv_orders(country VARCHAR(20) UNIQUE VALUES ('Germany', 'Switzerland', 'Belgium'),
customer_id INTEGER,
order_number BIGINT,
site VARCHAR(20) UNIQUE VALUES ('Prebook', 'At Once'),
season DATE,
order_type VARCHAR(10) UNIQUE VALUES ('B2B'),
product_code VARCHAR(20),
sku VARCHAR(20),
quantity_sku BIGINT,
baseprice_sku FLOAT,
totalprice_sku FLOAT,
currency VARCHAR(10) UNIQUE VALUES ('EUR', 'CHF'),
region VARCHAR(50),
sales_rep_id INTEGER,
sales_rep_name VARCHAR(50),
size VARCHAR(10))
Foreign Key: nlv_orders(customer_id) references nlv_customers(customer_id), nlv_orders(product_code) references nlv_products(product_code)
