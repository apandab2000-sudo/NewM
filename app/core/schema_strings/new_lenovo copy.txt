	global_l0 VARCHAR NULL - Program Type Gross To Net (GTN) global L0, 
	global_l1 VARCHAR NULL - Program Type Gross To Net (GTN) global L1, 
	global_l2 VARCHAR NULL - Program Type Gross To Net (GTN) global L2, 
	top_seller VARCHAR NULL - Indicator for top-selling products, 
	si_type VARCHAR NULL, 
	E_SI FLOAT NULL - Expense Sell in, 
	SI_Gross_Revenue FLOAT NULL - Sell In Gross Revenue, 
	SI_Net_Revenue FLOAT NULL - Sell In Net Revenue, 
	payment_usd_fin FLOAT NULL, 
	payment_usd FLOAT NULL - Rebate GTN Payment USD, 
	GTN FLOAT NULL - Gross To Net (GTN), 
	ST_Revenue FLOAT NULL - Sell Through Revenue, 
	BOX INTEGER NULL - Quantity Unit, 
	gross_profit FLOAT NULL, 
	gross_profit_revenue FLOAT NULL - Gross Profit Revenue. This is used as denominator for GP%, 
	budget_usd FLOAT NULL, 
	D_Inv_Aged_60 INTEGER NULL, 
	R_Inv_Aged_60 INTEGER NULL, 
	Inv_Aged_60 INTEGER NULL - Inventory Aged 60, 
	D_OH_Inv INTEGER NULL, 
	R_OH_Inv INTEGER NULL, 
	OH_Inv INTEGER NULL - On Hand Inventory, 
	gtn_tact FLOAT NULL - Tactical Gross To Net (GTN), 
	gtn_contr FLOAT NULL - Contractual Gross To Net (GTN), 
	budget_tact FLOAT NULL, 
	budget_contr FLOAT NULL, 
	fiscal_year_quarter_end DATE NULL - Fiscal year and quarter (Eg., 2023-10-31)
